
  useEffect(() => {
    if (!threadId || !userId) return
    const fetchTyping = () => {
      fetch("/api/messages/typing?threadId=" + encodeURIComponent(threadId) + "&userId=" + encodeURIComponent(userId))
        .then((res) => res.json())
        .then((data) => {
          setIsOtherTyping(Array.isArray(data.typing) && data.typing.length > 0)
        })
        .catch(() => {
          setIsOtherTyping(false)
        })
    }
    fetchTyping()
    const interval = setInterval(fetchTyping, 2000)
    return () => {
      clearInterval(interval)
    }
  }, [threadId, userId])

  useEffect(() => {
    if (!threadId || !userId) return
    if (typingDebounceRef.current) clearTimeout(typingDebounceRef.current)
    typingDebounceRef.current = setTimeout(() => {
      fetch("/api/messages/typing", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ threadId, userId, isTyping }),
      }).catch(() => null)
    }, 300)
    return () => {
      if (typingDebounceRef.current) clearTimeout(typingDebounceRef.current)
    }
  }, [isTyping, threadId, userId])
